🔒 推薦的 Firestore 安全規則 (Security Rules)：

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /company_tracker/{document=**} {
      // 僅允許經由本系統驗證、且在授權名單內的信箱讀寫
      allow read, write: if request.auth != null;
    }
  }
}
